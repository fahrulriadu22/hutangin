# Firestore Security Rules - Hutangin

## Version
`rules_version = '2'`

## Complete Rules

```javascript
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // ============================================================
    // 1. USERS COLLECTION
    // Setiap user hanya bisa baca/tulis data mereka sendiri
    // ============================================================
    match /users/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
      allow create: if request.auth != null;
    }
    
    // ============================================================
    // 2. DEBTORS COLLECTION
    // User hanya bisa akses debitur yang mereka buat sendiri
    // ============================================================
    match /debtors/{debtorId} {
      allow read: if request.auth != null && 
        resource.data.userId == request.auth.uid;
      
      allow create: if request.auth != null && 
        request.resource.data.userId == request.auth.uid;
      
      allow update: if request.auth != null && 
        resource.data.userId == request.auth.uid &&
        (request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          'name', 'phone', 'email', 'ratingAvg', 'totalTransactions', 'updatedAt'
        ]));
      
      allow delete: if request.auth != null && 
        resource.data.userId == request.auth.uid &&
        resource.data.totalTransactions == 0; // Hanya bisa hapus jika belum punya transaksi
    }
    
    // ============================================================
    // 3. TRANSACTIONS ACTIVE COLLECTION
    // ============================================================
    match /transactions_active/{transactionId} {
      allow read: if request.auth != null && 
        resource.data.userId == request.auth.uid;
      
      allow create: if request.auth != null && 
        request.resource.data.userId == request.auth.uid &&
        request.resource.data.amount >= 10000 && // Minimal Rp 10.000
        request.resource.data.deadline > request.resource.data.createdAt; // Deadline harus masa depan
      
      allow update: if request.auth != null && 
        resource.data.userId == request.auth.uid &&
        resource.data.status != 'paid' && // Tidak bisa update jika sudah lunas
        (request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          'amount', 'interestRate', 'interestType', 'deadline', 
          'notes', 'status', 'reminderCount', 'lastReminderAt', 'updatedAt'
        ]));
      
      allow delete: if false; // Tidak boleh hapus, harus dipindah ke archived
    }
    
    // ============================================================
    // 4. TRANSACTIONS ARCHIVED COLLECTION
    // Read-only setelah dipindah
    // ============================================================
    match /transactions_archived/{transactionId} {
      allow read: if request.auth != null && 
        resource.data.userId == request.auth.uid;
      
      allow create: if request.auth != null && 
        request.resource.data.userId == request.auth.uid &&
        request.resource.data.status == 'paid' &&
        request.resource.data.paidAt != null;
      
      allow update: if false; // Tidak bisa update archived
      allow delete: if false; // Tidak bisa hapus archived
    }
    
    // ============================================================
    // 5. REMINDER LOGS COLLECTION
    // Auto-generated by cloud functions, user read-only
    // ============================================================
    match /reminder_logs/{logId} {
      allow read: if request.auth != null && 
        resource.data.userId == request.auth.uid;
      
      allow create: if false; // Hanya cloud function yang bisa create
      allow update: if false;
      allow delete: if false;
    }
    
    // ============================================================
    // 6. RATINGS COLLECTION
    // ============================================================
    match /ratings/{ratingId} {
      allow read: if request.auth != null && 
        resource.data.userId == request.auth.uid;
      
      allow create: if request.auth != null && 
        request.resource.data.userId == request.auth.uid &&
        request.resource.data.rating >= 1 && 
        request.resource.data.rating <= 5;
      
      allow update: if false; // Tidak bisa update rating
      allow delete: if false;
    }
    
    // ============================================================
    // 7. BLOCKED DEBTORS COLLECTION
    // ============================================================
    match /blocked_debtors/{blockId} {
      allow read: if request.auth != null && 
        resource.data.userId == request.auth.uid;
      
      allow create: if request.auth != null && 
        request.resource.data.userId == request.auth.uid;
      
      allow delete: if request.auth != null && 
        resource.data.userId == request.auth.uid;
    }
    
  }
}
